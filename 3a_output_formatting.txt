import requests
import json


def emotion_detector(text_to_analyze: str):
    url = "https://sn-watson-emotion.labs.skills.network/v1/watson.runtime.nlp.v1/NlpService/EmotionPredict"
    headers = {"grpc-metadata-mm-model-id": "emotion_aggregated-workflow_lang_en_stock"}
    input_json = {"raw_document": {"text": text_to_analyze}}
    response = requests.post(url=url, headers=headers, json=input_json)

    json_out = json.loads(response.text)
    emotion_scores = json_out["emotionPredictions"][0]["emotion"]
    highest_emotion_score = max(emotion_scores, key=emotion_scores.get)

    emotion_scores["dominant_emotion"] = highest_emotion_score

    return emotion_scores
